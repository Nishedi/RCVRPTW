\end{equation}

\begin{equation}{\label{eq:goal_work_time}} %czas pracy
    \sum_{v \in \mathcal{V}} \left(\sum_{i\in\mathcal{L}}\sum_{j\in\mathcal{L}}x_{v,i,j}d_{i,j}
    +\sum_{i\in\mathcal{D}}w_{v,i}+\sum_{i\in\mathcal{D}}s_iz_{v,i}
    -T^l_{v}
    \right)
\end{equation}

\begin{equation}{\label{eq:penalty_function}}
     \sum_{v \in \mathcal{V}}\sum_{i \in \mathcal{D}}( \alpha (p_{v,i} + q_{v,i}))
\end{equation}

\begin{equation}
     x_{v,i,j}, z_{v,i}\in\{0,1\}
\end{equation}

\begin{equation}
     p_{v,i}, q_{v,i}, w_{v,i}, A_{v,i},S_{v,i},d_{i,j} \geq 0
\end{equation}

\begin{equation}{\label{eq:one_visit}}
     \sum_{v \in \mathcal{V}}\sum_{j \in \mathcal{D}\setminus i} x_{v,i,j}=1, \forall i \in \mathcal{D}
\end{equation}

\begin{equation}{\label{eq:flow_bilans}}
    \sum_{j\neq i}x_{v,i,j}-\sum_{j\neq i}x_{v,i,j}=0 \hspace{1cm} \forall v,i \in\mathcal{D}\setminus0
\end{equation}

\begin{equation}{\label{eq:start_end_in_depot}}
    \sum_{j\neq 0}x_{v,0,j}=z_v \space \space \sum_{i\neq 0}x_{v,i,0}=z_v \space \forall v
\end{equation}

\begin{equation}{\label{eq:edgevisitcorrelation}}
    z_{v,i} = \sum_{j\in \mathcal{D}\setminus{i}}x_{v,j,i} \hspace{1cm} z_{v,i} = \sum_{j\in \mathcal{D}\setminus{i}}x_{v,i,j}
\end{equation}

% 4) Spójność czasowa (przejazd + serwis -> przyjazd)
\begin{equation}{\label{eq:time_cohestion}}
 A_{v,j} \;\ge\; S_{v,i} + s_i + t_{ij} \;-\; M(1 - x_{v,i,j}), 
\quad \forall v,\; \forall i\neq j. \tag{4a}   
\end{equation}

\begin{equation}\label{eq:start_soft}
S_{v,i} \;\ge\; A_{v,i}, \qquad a_i \;\le\; S_{v,i} + p_{v,i}, \quad p_{v,i}\ge 0, \quad \forall v,i. \tag{4b'}
\end{equation}

\begin{equation}\label{eq:waiting_time}
w_{v,i} \;=\; S_{v,i} - A_{v,i}, \quad w_{v,i}\ge 0, \quad \forall v,i.
\end{equation}
\begin{equation}\label{eq:penalty}
S_{v,i} \;\le\; b_i + q_{v,i}, \quad q_{v,i}\ge 0, \quad \forall v,i.
\end{equation}
\begin{equation}{\label{eq:notimesforvehiclenotvisit}}
    A_{v,i}\leq a_i + M \sum_j x_{v,j,i}, \hspace{1cm} S_{v,i}\leq b_i
+M \sum_j x_{v,j,i}
\end{equation}

\begin{equation}\label{eq:capacity_flow}
y_{v,j} \;\ge\; y_{v,i} + q_j - Q_v(1 - x_{v,i,j}), 
\quad \forall v \in \mathcal{V},\; i,j \in \mathcal{D},\; i \neq j.
\end{equation}

\begin{equation}\label{eq:capacity_limits}
0 \;\le\; y_{v,i} \;\le\; Q_v z_{v,i}, 
\quad \forall v \in \mathcal{V},\; i \in \mathcal{D}\setminus\{0\}, 
\qquad y_{v,0} = 0,\; \forall v.
\end{equation}



\ref{eq:one_visit} Każdy lokalizacja jest odwiedzana dokładnie jeden raz.
\ref{eq:flow_bilans} Jeśli pojazd $v$ przejeżdża od klienta $i$ do $j$, to musi też z niego wyjechać do innego klienta.
\ref{eq:start_end_in_depot} Każdy pojazd, który wyjeżdża do obsługi musi wyjechać z bazy i do niej wrócić
\ref{eq:time_cohestion} Przyjazd do kolejnego miasta nie może być wcześniejszy niż suma czasu rozpoczęcia serwisu, czasu wykonywania serwisu oraz czasu przejazdu z $i$ do $j$.
\ref{eq:start_soft} Czas rozpoczęcia serwisu, musi być większy bądź równy jak czas przyjazdu do lokalizacji. 
\ref{eq:waiting_time} Czas oczekiwania jest równy różnicy miedzy czasem rozpoczęcia serwisu, a przyjazdem do lokalizacji
\ref{eq:edgevisitcorrelation}.
\ref{eq:notimesforvehiclenotvisit} Wyłączenie czasów lokalizacji nieodwiedzanych przez pojazd $v$.
\ref{eq:capacity_flow} przy każdym przejeździe ładunek rośnie o popyt lokalizacji.
\ref{eq:capacity_limits} Ładunek pojazdu musi być nieujemny i  aktualne załadowanie musi być mniejsze niż maksymalna pojemność pojazdu, jeżeli lokalizacja jest odwiedzana.

\subsection{Definition of variables}
In problem formulated we assume the existence of $n$ service locations in set $\mathcal{N} = {1, 2, \dots, N}$. This set is extended by location 0, representing the starting node called depot, creating a set $\mathcal{L} = {0} \cup \mathcal{N}$. Depot is a start and end point for each of service vehicles. Each location has defined its coordinates, based on which, the cost travel between locations is calculated. The cost is represented by the matrix 


Finaly, the constraints for each maintenance vehicle are defined by $C \in [0, 1000]$ and $T \in [0, 480]$, which are respectively capacity and working time limit constraints. The maximal capactity is set to 1000 kg, as it is an average load for small and medium delivery van %powołać się na ładowności ducato.
The time constraint is set to 480 minutes, which is in line with the 8-hour working day in force in Poland.
Additionaly we introduce waiting time, as it is needed to pay the employees even if they are out of work.

\subsection{Model formulation}
The model of given Heterogenous Capacitated Vehicle Routing Problem with Time Windows (HCVRP-TW) consists of two parts. As in this paper Mixed Integer Linear Programming (MILP) model is used, the first element defined is a goal function $F(\pi) = F_1(\pi) + F_1(\pi)$ that minimizes the total cost of work performed by all teams. This function must be considered taking into account the constraints of the problem defined as the second part of the model.
